//Sum of each Row
#include<iostream>
using namespace std;
int main()
{
	int arr[3][3] = { {1,2,3}, 
	                  {4,5,6}, 
		          {7,8,9} };
	int sum = 0;
	for (int i = 0; i < 3; i++)
	{
		sum = 0;
		for (int j = 0; j < 3; j++)
		{
			cout << arr[i][j] << " ";
			sum += arr[i][j];
		}
		cout<<"= "<<sum << endl;
	}
	system("pause");
}

//array in functions
void func2D(int a[][3])
{
	for (int i = 0; i < 3; i++)
	{
		for (int j = 0; j < 3; j++)
		{
			cout << a[i][j] << "  ";
		}
		cout << endl;
	}
}
int main()
{
	int arr[3][3] = { {1,2,3}, 
	                  {4,5,6}, 
					  {7,8,9} };

}
