#include<iostream>
#include<fstream>
using namespace std;
#define size 20
int main()
{
	char arr[size];
	ofstream fout;
	char filename[] = "charater_array.txt";
	fout.open(filename);
	cout << "Enter C-String: ";
	cin >> arr;
	fout << arr;
	fout.close();

	ifstream fin;
	fin.open(filename);
	char r[size] = { '\0' };
	int i = 0;
	while (!fin.eof())
	{
		fin >> r[i];
		i++;
	}
	cout << "\nFrom File: " << r;
	fin.close();	
}
