int main(){
	int count = 0;
	char arr[][3] = { { 'a', 'b', 'c' },
	                  { 'a', 'b', 'z' } };
	for (int i = 0; i < 1; i++)
	{
		for (int j = 0; j < 3; j++)
		{
			if (arr[i][j] == arr[i + 1][j])
				count++;
		}
	}
	cout << count;
}

int main(){
	char arr[][3] = { { 'a', 'b', 'C' },
	                  { 'a', 'B', 'z' } };
	for (int i = 0; i < 1; i++)
	{
		for (int j = 0; j < 3; j++)
		{
			if (arr[i][j] >= 'a' && arr[i][j] <= 'z')
				arr[i][j] = arr[i][j] - 32;
			else
				arr[i][j] = arr[i][j] - 32;
		}
	}
}
